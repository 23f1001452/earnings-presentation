Sample: Financial Data Extraction (Python)


# Compute revenue from transactions
def compute_revenue(transactions):
    # transactions: list of dicts with keys 'qty' and 'price'
    return sum(t['qty'] * t['price'] for t in transactions)

# Example
transactions = [{'qty': 10, 'price': 100.0}, {'qty': 5, 'price': 250.0}]
print(compute_revenue(transactions))  # 2250.0
